        item = produce_item();

    }
}

9. Producer-Consumer Solution using Java Monitors

    Slide Context: High-level monitor implementation for multi-threaded thread synchronization.   
    PDF+ 1

Java

public class ProducerConsumer {
    static final int N = 100; // buffer size
    static producer p = new producer();
    static consumer c = new consumer();
    static our_monitor mon = new our_monitor();

    public static void main(String args[]) {
        p.start();
        c.start();
    }

    static class producer extends Thread {
        public void run() {
            int item;
            while (true) {
                item = produce_item();
                mon.insert(item);
            }
        }
        private int produce_item() { /* generate item */ return 0; }
    }

    static class consumer extends Thread {
        public void run() {
            int item;
            while (true) {
                item = mon.remove();
                consume_item(item);
            }
        }
        private void consume_item(int item) { /* consume item */ }
    }

    static class our_monitor { // Monitor definition
        private int buffer[] = new int[N];
        private int count = 0, lo = 0, hi = 0;

        public synchronized void insert(int val) {
            if (count == N) go_to_sleep();
            buffer[hi] = val;
            hi = (hi + 1) % N;
            count = count + 1;
            if (count == 1) notify();
        }

        public synchronized int remove() {
            int val;
            if (count == 0) go_to_sleep();
            val = buffer[lo];
            lo = (lo + 1) % N;
            count = count - 1;
            if (count == N - 1) notify();
            return val;
        }

        private void go_to_sleep() {
            try { wait(); } catch (InterruptedException exc) {}
        }
    }
}

10. Producer-Consumer with Message Passing

    Slide Context: IPC implementation suitable for distributed systems without shared memory.   
    PDF

C

#define N 100 /* number of slots in the buffer */

void producer(void) {
    int item;
    message m;
    while (TRUE) {
        item = produce_item();
        receive(consumer, &m);      /* wait for an empty reply to arrive */
        build_message(&m, item);    /* construct message */
        send(consumer, &m);         /* send item to consumer */
    }
}

void consumer(void) {
    int item, i;
    message m;
    
    for (i = 0; i < N; i++) send(producer, &m); /* send N empty messages */
    
    while (TRUE) {
        receive(producer, &m);      /* get message containing item */
        item = extract_item(&m);    /* extract item */
        send(producer, &m);         /* send back empty reply */
        consume_item(item);         /* consume item */
    }
}

11.3 State-based Optimal Solution (Deadlock-Free & Parallelized)   
C

#define N           5                   /* number of philosophers */
#define LEFT        (i + N - 1) % N     /* number of i's left neighbor */
#define RIGHT       (i + 1) % N         /* number of i's right neighbor */
#define THINKING    0                   /* philosopher is thinking */
#define HUNGRY      1                   /* philosopher is trying to get forks */
#define EATING      2                   /* philosopher is eating */

typedef int semaphore;

int state[N];                           /* array to keep track of everyone's state */
semaphore mutex = 1;                    /* mutual exclusion for critical regions */
semaphore s[N];                         /* one semaphore per philosopher */

void philosopher(int i) {
    while (TRUE) {
        think();
        take_forks(i);                  /* acquire two forks or block */
        eat();
        put_forks(i);                   /* put both forks back on table */
    }
}

void take_forks(int i) {
    down(&mutex);                       /* enter critical region */
    state[i] = HUNGRY;
    test(i);                            /* try to acquire 2 forks */
    up(&mutex);                         /* exit critical region */
    down(&s[i]);                        /* block if forks were not acquired */
}

void put_forks(int i) {
    down(&mutex);                       /* enter critical region */
    state[i] = THINKING;                /* philosopher finished eating */
    test(LEFT);                         /* see if left neighbor can now eat */
    test(RIGHT);                        /* see if right neighbor can now eat */
    up(&mutex);                         /* exit critical region */
}

void test(int i) {
    if (state[i] == HUNGRY && state[LEFT] != EATING && state[RIGHT] != EATING) {
        state[i] = EATING;
        up(&s[i]);
    }
}

12. Readers and Writers Problem (First Variation - Reader Priority)

    Slide Context: Database concurrency control prioritizing simultaneous reader processes.   
    PDF

C

typedef int semaphore;
semaphore mutex = 1;    /* controls access to 'rc' */
semaphore db = 1;       /* controls access to database */
int rc = 0;             /* number of processes reading or wanting to */

void reader(void) {
    while (TRUE) {
        down(&mutex);
        rc = rc + 1;
        if (rc == 1) down(&db);  /* first reader locks database */
        up(&mutex);
        
        read_data_base();
        
        down(&mutex);
        rc = rc - 1;
        if (rc == 0) up(&db);    /* last reader unlocks database */
        up(&mutex);
        
        use_data_read();
    }
}

void writer(void) {
    while (TRUE) {
        think_up_data();
        
        down(&db);              /* get exclusive access to DB */
        write_data_base();
        up(&db);                /* release exclusive access to DB */
    }
}
